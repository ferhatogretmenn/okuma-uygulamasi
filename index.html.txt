let currentGroupIdx = 0;
let selectedLetter = null;
let availableLetters = [];
let wordPool = [];
let currentWord = "";
let score = 0;

/* ===== DOM ===== */
const groupsEl = document.getElementById('groups');
const lettersEl = document.getElementById('letters');
const wordEl = document.getElementById('word');
const scoreEl = document.getElementById('score');

/* ===== BAŞLAT ===== */
function init() {
  GROUPS.forEach((g, i) => {
    const b = document.createElement('button');
    b.className = 'group-btn' + (i === 0 ? ' active' : '');
    b.textContent = g.name;
    b.onclick = () => selectGroup(i);
    groupsEl.appendChild(b);
  });
  selectGroup(0);
}

function selectGroup(idx) {
  currentGroupIdx = idx;
  document.querySelectorAll('.group-btn').forEach((b, i) => {
    b.classList.toggle('active', i === idx);
  });

  // O gruba kadar olan tüm harfler kullanılabilir
  availableLetters = [];
  for (let i = 0; i <= idx; i++) availableLetters.push(...GROUPS[i].letters);

  // O grubun harfleri gösterilir
  lettersEl.innerHTML = '';
  GROUPS[idx].letters.forEach(l => {
    const b = document.createElement('button');
    b.className = 'letter-btn';
    b.textContent = l;
    b.onclick = () => selectLetter(l, b);
    lettersEl.appendChild(b);
  });

  selectedLetter = null;
  wordEl.textContent = '?';
}

function selectLetter(letter, btn) {
  selectedLetter = letter;
  document.querySelectorAll('.letter-btn').forEach(b => b.classList.remove('active'));
  btn.classList.add('active');

  // O harfe kadar öğrenilen harflerden oluşan kelimeleri filtrele
  wordPool = ALL_WORDS.filter(w => {
    if (!w.toLowerCase().includes(letter)) return false;
    // Kelimenin TÜM harfleri öğrenilmiş harfler arasında olmalı
    return [...w.toLowerCase()].every(ch => availableLetters.includes(ch));
  });

  // Tekrarı önlemek için karıştır
  wordPool = shuffle([...wordPool]);

  if (wordPool.length === 0) {
    wordEl.textContent = '😕';
    wordEl.classList.remove('animate');
    return;
  }
  nextWord();
}

function nextWord() {
  if (wordPool.length === 0) return;
  currentWord = wordPool[Math.floor(Math.random() * wordPool.length)];
  wordEl.textContent = currentWord;
  wordEl.classList.remove('animate');
  void wordEl.offsetWidth;
  wordEl.classList.add('animate');
}

function speakWord() {
  if (!currentWord) return;
  if ('speechSynthesis' in window) {
    speechSynthesis.cancel();
    const u = new SpeechSynthesisUtterance(currentWord);
    u.lang = 'tr-TR';
    u.rate = 0.8;
    u.pitch = 1.1;
    speechSynthesis.speak(u);
  }
}

function starIt() {
  score++;
  scoreEl.textContent = score;
  burstConfetti();

  // Tebrik sesi (kısa yüksek ton)
  if ('speechSynthesis' in window) {
    speechSynthesis.cancel();
    const u = new SpeechSynthesisUtterance('Aferin!');
    u.lang = 'tr-TR';
    u.rate = 1.1;
    u.pitch = 1.3;
    speechSynthesis.speak(u);
  }
  setTimeout(nextWord, 700);
}

function burstConfetti() {
  const emojis = ['⭐','🎉','🌟','✨','🎈','🏆','💫'];
  for (let i = 0; i < 15; i++) {
    const c = document.createElement('div');
    c.className = 'confetti';
    c.textContent = emojis[Math.floor(Math.random()*emojis.length)];
    c.style.left = Math.random() * 100 + 'vw';
    c.style.animationDelay = Math.random() * 0.3 + 's';
    document.body.appendChild(c);
    setTimeout(() => c.remove(), 2500);
  }
}

function shuffle(arr) {
  for (let i = arr.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
  return arr;
}
